<body>
  <main>
    <h1>PONG</h1>
    <div class="scoreboard" aria-label="Score">
      <div>You: <span class="score" id="player-score">0</span></div>
      <div>Computer: <span class="score" id="computer-score">0</span></div>
    </div>

    <canvas id="game" width="800" height="500"
      aria-label="Pong game. Move your paddle with the mouse or up and down arrow keys."></canvas>

    <p>Move your mouse over the game or use the ↑ and ↓ arrow keys.</p>
  </main>

  <script>
    const canvas = document.getElementById("game");
    const ctx = canvas.getContext("2d");
    const playerScoreElement = document.getElementById("player-score");
    const computerScoreElement = document.getElementById("computer-score");

    const WIDTH = canvas.width;
    const HEIGHT = canvas.height;
    const PADDLE_WIDTH = 12;
    const PADDLE_HEIGHT = 90;
    const BALL_RADIUS = 9;
    const PLAYER_SPEED = 420;
    const COMPUTER_SPEED = 310;

    const player = { x: 28, y: (HEIGHT - PADDLE_HEIGHT) / 2 };
    const computer = { x: WIDTH - 28 - PADDLE_WIDTH, y: (HEIGHT - PADDLE_HEIGHT) / 2 };
    const ball = { x: WIDTH / 2, y: HEIGHT / 2, vx: 300, vy: 150 };

    const keys = new Set();
    let playerScore = 0;
    let computerScore = 0;
    let previousTime = 0;

    function clampPaddle(paddle) {
      paddle.y = Math.max(0, Math.min(HEIGHT - PADDLE_HEIGHT, paddle.y));
    }

    function movePlayerToPointer(event) {
      const rect = canvas.getBoundingClientRect();
      const scaleY = HEIGHT / rect.height;
      player.y = (event.clientY - rect.top) * scaleY - PADDLE_HEIGHT / 2;
      clampPaddle(player);
    }

    canvas.addEventListener("pointermove", movePlayerToPointer);

    window.addEventListener("keydown", (event) => {
      if (event.key === "ArrowUp" || event.key === "ArrowDown") {
        event.preventDefault();
        keys.add(event.key);
      }
    });

    window.addEventListener("keyup", (event) => keys.delete(event.key));
    window.addEventListener("blur", () => keys.clear());

    function resetBall(direction) {
      ball.x = WIDTH / 2;
      ball.y = HEIGHT / 2;
      ball.vx = 300 * direction;
      ball.vy = (Math.random() * 240) - 120;
    }

    function hitPaddle(paddle, direction) {
      const overlapsY =
        ball.y + BALL_RADIUS > paddle.y &&
        ball.y - BALL_RADIUS < paddle.y + PADDLE_HEIGHT;

      if (!overlapsY) return false;

      if (direction > 0) {
        return ball.vx < 0 &&
          ball.x - BALL_RADIUS <= paddle.x + PADDLE_WIDTH &&
          ball.x > paddle.x;
      }

      return ball.vx > 0 &&
        ball.x + BALL_RADIUS >= paddle.x &&
        ball.x < paddle.x + PADDLE_WIDTH;
    }

    function bounceOffPaddle(paddle, direction) {
      const paddleCenter = paddle.y + PADDLE_HEIGHT / 2;
      const impact = (ball.y - paddleCenter) / (PADDLE_HEIGHT / 2);

      ball.vx = direction * Math.min(Math.abs(ball.vx) * 1.06, 620);
      ball.vy = impact * 360;

      // Place the ball just outside the paddle to prevent repeated collisions.
      ball.x = direction > 0
        ? paddle.x + PADDLE_WIDTH + BALL_RADIUS
        : paddle.x - BALL_RADIUS;
    }

    function update(deltaTime) {
      const dt = Math.min(deltaTime, 0.03);

      if (keys.has("ArrowUp")) player.y -= PLAYER_SPEED * dt;
      if (keys.has("ArrowDown")) player.y += PLAYER_SPEED * dt;
      clampPaddle(player);

      // The computer tracks the ball, but moves at a limited speed.
      const targetY = ball.vx > 0
        ? ball.y - PADDLE_HEIGHT / 2
        : HEIGHT / 2 - PADDLE_HEIGHT / 2;
      const difference = targetY - computer.y;
      const movement = Math.min(Math.abs(difference), COMPUTER_SPEED * dt);
      computer.y += Math.sign(difference) * movement;
      clampPaddle(computer);

      ball.x += ball.vx * dt;
      ball.y += ball.vy * dt;

      // Top and bottom wall collisions.
      if (ball.y - BALL_RADIUS <= 0) {
        ball.y = BALL_RADIUS;
        ball.vy = Math.abs(ball.vy);
      } else if (ball.y + BALL_RADIUS >= HEIGHT) {
        ball.y = HEIGHT - BALL_RADIUS;
        ball.vy = -Math.abs(ball.vy);
      }

      if (hitPaddle(player, 1)) bounceOffPaddle(player, 1);
      if (hitPaddle(computer, -1)) bounceOffPaddle(computer, -1);

      // A point is scored when the ball passes either paddle.
      if (ball.x + BALL_RADIUS < 0) {
        computerScore++;
        computerScoreElement.textContent = computerScore;
        resetBall(1);
      } else if (ball.x - BALL_RADIUS > WIDTH) {
        playerScore++;
        playerScoreElement.textContent = playerScore;
        resetBall(-1);
      }
    }

    function draw() {
      ctx.clearRect(0, 0, WIDTH, HEIGHT);

      // Dashed center line.
      ctx.strokeStyle = "#414a61";
      ctx.lineWidth = 3;
      ctx.setLineDash([10, 12]);
      ctx.beginPath();
      ctx.moveTo(WIDTH / 2, 0);
      ctx.lineTo(WIDTH / 2, HEIGHT);
      ctx.stroke();
      ctx.setLineDash([]);

      ctx.fillStyle = "#f4f6ff";
      ctx.fillRect(player.x, player.y, PADDLE_WIDTH, PADDLE_HEIGHT);
      ctx.fillRect(computer.x, computer.y, PADDLE_WIDTH, PADDLE_HEIGHT);

      ctx.beginPath();
      ctx.arc(ball.x, ball.y, BALL_RADIUS, 0, Math.PI * 2);
      ctx.fill();
    }

    function frame(time) {
      const deltaTime = previousTime ? (time - previousTime) / 1000 : 0;
      previousTime = time;

      update(deltaTime);
      draw();
      requestAnimationFrame(frame);
    }

    resetBall(Math.random() < 0.5 ? -1 : 1);
    requestAnimationFrame(frame);
  </script>
